def S031 : Prop :=
  ∀ (n : ℕ) (A : Op n) (ψ : Config n → ℂ), energy A ψ = energyOn A ψ

--- QPCP.Config (QPCP/Defs.lean) ---
abbrev Config (n : ℕ) : Type := Fin n → Fin 2

--- QPCP.Op (QPCP/Defs.lean) ---
abbrev Op (n : ℕ) : Type := Matrix (Config n) (Config n) ℂ

--- QPCP.energy (QPCP/Defs.lean) ---
def energy {n : ℕ} (A : Op n) (ψ : Config n → ℂ) : ℝ :=
  (dotProduct (star ψ) (A.mulVec ψ)).re

--- QPCP.energyOn (QPCP/Qudit.lean) ---
def energyOn {ι : Type} [Fintype ι] (A : Matrix ι ι ℂ) (ψ : ι → ℂ) : ℝ :=
  (dotProduct (star ψ) (A.mulVec ψ)).re
